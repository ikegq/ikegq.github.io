### canvas 绘制和拖动

```html
<!DOCTYPE html>
<html lang="en">

<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Document</title>

  <style>
    body {
      display: flex;
      flex-direction: column;
      align-items: center;
    }

    canvas {
      /* width: 500px;
      height: 300px; */
      border: 1px solid #000;
    }
  </style>


</head>

<body>
  <input type="color" />
  <canvas></canvas>

  <script>
    const canvas = document.querySelector('canvas');
    const ctx = canvas.getContext('2d');
    const colorInput = document.querySelector('input[type="color"]');

    function init() {
      const x = 500, y = 300
      canvas.width = x * devicePixelRatio
      canvas.height = y * devicePixelRatio
      canvas.style.width = x + 'px'
      canvas.style.height = y + 'px'
    }

    init()

    const shapes = []

    class Rectangle {
      constructor(startX, startY, color) {
        this.startX = startX
        this.startY = startY
        this.color = color
        this.endX = startX
        this.endY = startY
      }

      draw() {
        ctx.fillStyle = this.color
        ctx.strokeStyle = 'red'
        ctx.lineWidth = 3 * devicePixelRatio
        ctx.fillRect(this.minX * devicePixelRatio, this.minY * devicePixelRatio, (this.maxX - this.minX) * devicePixelRatio, (this.maxY - this.minY) * devicePixelRatio)
        ctx.strokeRect(this.minX * devicePixelRatio, this.minY * devicePixelRatio, (this.maxX - this.minX) * devicePixelRatio, (this.maxY - this.minY) * devicePixelRatio)

      }

      get minX() {
        return Math.min(this.startX, this.endX)
      }

      get maxX() {
        return Math.max(this.startX, this.endX)
      }

      get minY() {
        return Math.min(this.startY, this.endY)
      }

      get maxY() {
        return Math.max(this.startY, this.endY)
      }

      isInside(x, y) {
        return x >= this.minX && x <= this.maxX && y >= this.minY && y <= this.maxY
      }
    }

    function getShape(x, y) {
      for (let i = shapes.length - 1; i >= 0; i--) {
        if (shapes[i].isInside(x, y)) {
          return shapes[i]
        }
      }
      return null
    }

    canvas.onmousedown = e => {
      let shape = getShape(e.offsetX, e.offsetY)
      if (shape) {
        const cvs = canvas.getBoundingClientRect()
        let startX = shape.startX, startY = shape.startY, endX = shape.endX, endY = shape.endY
        window.onmousemove = e2 => {
          let x = e2.offsetX - e.offsetX, y = e2.offsetY - e.offsetY
          shape.startX = startX + x
          shape.startY = startY + y
          shape.endX = endX + x
          shape.endY = endY + y
        }
        window.onmouseup = e => {
          window.onmousemove = null
          window.onmouseup = null
        }

      } else {
        const rect = new Rectangle(e.offsetX, e.offsetY, colorInput.value)
        shapes.push(rect)

        window.onmousemove = e2 => {
          rect.endX = e2.offsetX
          rect.endY = e2.offsetY
        }
        window.onmouseup = e => {
          window.onmousemove = null
          window.onmouseup = null
        }
      }
    }

    function draw() {
      requestAnimationFrame(draw)
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      shapes.forEach(shape => {
        shape.draw()
      })
    }

    draw()

  </script>
</body>

</html>
```
