```html
<!DOCTYPE html>
<html lang="en">

<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Document</title>
  <style>
    * {
      margin: 0;
      padding: 0;
    }

    canvas {
      width: 100vw;
      height: 100vh;
      display: block;
      background: radial-gradient(#fff, #8c738c);
    }
  </style>
</head>

<body>
  <canvas></canvas>
</body>

<script>
  const canvas = document.querySelector('canvas')
  const ctx = canvas.getContext('2d', {
    willReadFrequently: true
  })

  const initCanvas = () => {
    canvas.width = window.innerWidth * devicePixelRatio
    canvas.height = window.innerHeight * devicePixelRatio
  }

  initCanvas()

  const getRandom = (min, max) => {
    return Math.floor(Math.random() * (max - min + 1) + min)
  }

  class Particle {
    constructor() {
      this.size = getRandom(2 * devicePixelRatio, 6 * devicePixelRatio)
      const r = Math.min(canvas.width, canvas.height) / 2
      const rad = getRandom(0, 360) * Math.PI / 180

      this.x = canvas.width / 2 + r * Math.cos(rad)
      this.y = canvas.height / 2 + r * Math.sin(rad)
    }

    draw() {
      ctx.beginPath()
      ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2)
      ctx.fillStyle = '#5445544d'
      ctx.fill()
    }

    moveTo(tx, ty) {
      const duration = 500
      const sx = this.x, sy = this.y

      const xSpeed = (tx - sx) / duration
      const ySpeed = (ty - sy) / duration

      const startTime = Date.now()

      const _move = () => {
        const t = Date.now() - startTime

        const x = sx + xSpeed * t
        const y = sy + ySpeed * t

        this.x = x
        this.y = y
        if (t > duration) {
          this.x = tx
          this.y = ty
          return
        }

        requestAnimationFrame(_move)
      }

      _move()
    }
  }

  const clear = () => {
    ctx.clearRect(0, 0, canvas.width, canvas.height)
  }


  const getPoints = () => {
    const points = []

    const { data } = ctx.getImageData(0, 0, canvas.width, canvas.height)

    const gap = 5

    for (let i = 0; i < canvas.width; i += gap) {
      for (let j = 0; j < canvas.height; j += gap) {
        const index = (i + j * canvas.width) * 4
        const r = data[index]
        const g = data[index + 1]
        const b = data[index + 2]
        const a = data[index + 3]
        if (r === 0 && g === 0 && b === 0 && a === 255) {
          points.push([i, j])
        }
      }
    }

    return points
  }

  const update = () => {
    const curTimeText = new Date().toTimeString().substring(0, 8)
    if (timeText === curTimeText) {
      return
    }

    timeText = curTimeText

    const { width, height } = canvas

    ctx.fillStyle = '#000'
    ctx.textBaseline = 'middle'
    ctx.font = `${140 * devicePixelRatio}px 'DS-Digital', sans-serif`
    ctx.textAlign = 'center'
    ctx.fillText(timeText, width / 2, height / 2)

    const points = getPoints()

    clear()

    for (let i = 0; i < points.length; i++) {

      const [x, y] = points[i]
      let p = particles[i]

      if (!p) {
        p = new Particle()
        particles.push(p)
      }

      p.moveTo(x, y)
    }

    if (points.length < particles.length) {
      particles.splice(points.length)
    }

  }


  const particles = []
  let timeText = ''

  const draw = () => {
    clear()

    update()

    for (const p of particles) {
      p.draw()
    }

    requestAnimationFrame(draw)
  }

  draw()
</script>

</html>
```
