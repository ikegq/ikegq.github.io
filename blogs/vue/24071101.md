---
title: vue3 echarts tooltip formatter使用组件
date: 2024/07/11
tags:
  - vue
---

```vue
<script setup>
import { renderToString } from 'vue/server-renderer';
import CustomTooltip from './CustomTooltip.vue';

const myTooltipFormatter = async data => {
  const htmlTemplate = await renderToString(createApp(CustomTooltip, { data }));
  return htmlTemplate;
};

// echarts options
const options = {
  tooltip: {
    formatter: (params, ticket, callback) => {
      if (params.data) {
        myTooltipFormatter(params.data).then(result => {
          callback(ticket, result);
        });
        return '计算中...';
      }
    },
  },
};
</script>
```

```vue
<script setup>
const props = defineProps({
  data: {
    type: Object,
    default: () => ({}),
  },
});
</script>

<template>
  <div class="custom-tooltip"></div>
</template>
```

> 注意：引入的组件无法绑定事件
